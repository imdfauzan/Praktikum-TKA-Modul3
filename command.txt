# download multipass manual: https://cloud-images.ubuntu.com/noble/current/ pilih file `noble-server-cloudimg-amd64.img`
# rename file itu jadi `ubuntu.img` lalu buat folder Images di C:/

# buka terminal powershell baru

# Buka FIREWALL di PC node

netsh interface portproxy add v4tov4 listenaddress=0.0.0.0 listenport=2223 connectaddress=172.26.151.154 connectport=22
